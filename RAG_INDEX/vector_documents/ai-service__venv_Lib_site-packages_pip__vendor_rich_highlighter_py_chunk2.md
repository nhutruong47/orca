    """

    base_style = "iso8601."

        #
        # Dates

        # Calendar month (e.g. 2008-08). The hyphen is required

        # Calendar date w/o hyphens (e.g. 20080830)
        r"^(?P<date>(?P<year>[0-9]{4})(?P<month>1[0-2]|0[1-9])(?P<day>3[01]|0[1-9]|[12][0-9]))$",
