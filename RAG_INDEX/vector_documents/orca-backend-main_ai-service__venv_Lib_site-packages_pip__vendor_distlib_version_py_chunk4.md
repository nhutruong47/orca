    def parse(self, s):

        # clause, since that's needed to ensure that X.Y == X.Y.0 == X.Y.0.0

        groups = m.groups()
        self._release_clause = tuple(int(v) for v in groups[1].split('.'))
        return result

def _match_prefix(x, y):
    x = str(x)
    y = str(y)
    if x == y:
        return True
    if not x.startswith(y):
        return False
    n = len(y)
    return x[n] == '.'
