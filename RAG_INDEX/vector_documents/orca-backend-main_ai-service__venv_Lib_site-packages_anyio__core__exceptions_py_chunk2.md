class TaskFailed(Exception):
    """
    Raised when awaiting on, or attempting to access the return value of, a
    :class:`.TaskHandle` that raised an exception.
    """

    """
    Raised when awaiting on, or attempting to access the return value of, a
    :class:`.TaskHandle` that was cancelled.
    """
